def sigmoid(z):
    return 1 / (1 + np.exp(-z))
In [26]:
def grad_descent(X, y, lr=0.01, epochs = 10000):
    m,n = X.shape
    w = np.zeros(n)

    for i in range(epochs):
        z = X @ w
        h = sigmoid(z)
        gradient = (1/m) * (X.T @ (h - y))
        w -= lr * gradient

    return w
